local function getChar(plr)
	local c = plr.Character
	local hum = c and c:FindFirstChildOfClass("Humanoid")
	local hrp = c and c:FindFirstChild("HumanoidRootPart")
	if hum and hrp and hum.Health > 0 then return c, hum, hrp end
end

-- ===== roles =====
local function toolMatch(container, patterns)
	if not container then return false end
	for _, t in ipairs(container:GetChildren()) do
		if t:IsA("Tool") then
			local n = t.Name:lower()
			for _, p in ipairs(patterns) do
				if n:find(p, 1, true) then return true end
			end
		end
	end
	return false
end

local function roleOf(plr)
	if CONFIG.RoleResolver then
		local ok, r = pcall(CONFIG.RoleResolver, plr)
		if ok and r then return r end
	end
	local c, bp = plr.Character, plr:FindFirstChildOfClass("Backpack")
	if toolMatch(c, CONFIG.MurderTools) or toolMatch(bp, CONFIG.MurderTools) then return "murder" end
	if toolMatch(c, CONFIG.SheriffTools) or toolMatch(bp, CONFIG.SheriffTools) then return "sheriff" end
	return nil
end

-- ===== ESP =====
local function clearEsp(plr)
	local e = esp[plr]
	if e then
		pcall(function() e.hl:Destroy() end)
		pcall(function() e.bb:Destroy() end)
		esp[plr] = nil
	end
end

local function applyEsp(plr, role)
	local c = plr.Character
	local head = c and c:FindFirstChild("Head")
	if not (c and head) then clearEsp(plr) return end
	local e = esp[plr]
	local color = role == "murder" and CONFIG.ColorMurder or CONFIG.ColorSheriff
	local text = role == "murder" and "MURDER" or "XERIFE"
	if e and (e.hl.Adornee ~= c or e.role ~= role) then clearEsp(plr) e = nil end
	if not e then
		local hl = Instance.new("Highlight")
		hl.Adornee = c
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.FillTransparency = 0.5
		hl.Parent = gui
		local bb = Instance.new("BillboardGui")
		bb.Adornee = head
		bb.AlwaysOnTop = true
		bb.Size = UDim2.fromOffset(120, 30)
		bb.StudsOffset = Vector3.new(0, 2.5, 0)
		bb.Parent = gui
		local tl = Instance.new("TextLabel")
		tl.Size = UDim2.fromScale(1, 1)
		tl.BackgroundTransparency = 1
		tl.Font = Enum.Font.GothamBold
		tl.TextSize = 14
		tl.TextStrokeTransparency = 0.3
		tl.Parent = bb
		e = { hl = hl, bb = bb, tl = tl, role = role }
		esp[plr] = e
	end
	e.hl.FillColor = color
	e.hl.OutlineColor = color
	e.tl.TextColor3 = color
	e.tl.Text = text .. " - " .. plr.DisplayName
end

	end

local function visible(part, targetChar)
	if not state.wall then return true end
	local cam = Workspace.CurrentCamera
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { lp.Character, targetChar }
	return Workspace:Raycast(cam.CFrame.Position, part.Position - cam.CFrame.Position, params) == nil
end

local function aimTarget()
	local cam = Workspace.CurrentCamera
	local center = cam.ViewportSize / 2
	local best, bestD = nil, CONFIG.AimFov
	local mode = aimMode
	if mode == "auto" then
		-- eu murder -> mira no xerife; eu xerife (ou inocente) -> mira no murder
		mode = (roleOf(lp) == "murder") and "xerife" or "murder"
	end
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp then
			local c, _, _ = getChar(plr)
			local part = c and (c:FindFirstChild(CONFIG.AimPart) or c:FindFirstChild("HumanoidRootPart"))
			if part then
				local role = roleOf(plr)
				local okMode = mode == "qualquer" or (mode == "murder" and role == "murder") or (mode == "xerife" and role == "sheriff")
				if okMode then
					local v, onScreen = cam:WorldToViewportPoint(part.Position)
					if onScreen then
						local d = (Vector2.new(v.X, v.Y) - center).Magnitude
						if d < bestD and visible(part, c) then best, bestD = part, d end
					end
				end
			end
		end
	end
	return best
end

track(RunService.RenderStepped:Connect(function()
	fovCircle.Visible = state.aim
	if not state.aim then return end
	local part = aimTarget()
	if part then
		local cam = Workspace.CurrentCamera
		cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(cam.CFrame.Position, part.Position), CONFIG.AimSmooth)
	end
end))

-- ===== pegar arma caida (vira xerife pelas regras do jogo) =====
local function matchesDrop(d)
	if not (d:IsA("BasePart") or d:IsA("Model") or d:IsA("Tool")) then return false end
	local n = d.Name:lower()
	for _, want in ipairs(CONFIG.GunDropNames) do
		if n == want then return true end
	end
	return false
end
local function dropPart(d)
	if d:IsA("BasePart") then return d end
	return d:FindFirstChildWhichIsA("BasePart", true)
end
for _, d in ipairs(Workspace:GetDescendants()) do
	if matchesDrop(d) then drops[d] = true end
end
track(Workspace.DescendantAdded:Connect(function(d) if matchesDrop(d) then drops[d] = true end end))
track(Workspace.DescendantRemoving:Connect(function(d) drops[d] = nil end))

local lastGrab = 0
task.spawn(function()
	while alive do
		if state.gun and os.clock() - lastGrab > 1.5 then
			local myRole = roleOf(lp)
			local _, _, hrp = getChar(lp)
			if hrp and myRole == nil then
				for d in pairs(drops) do
					local part = d.Parent and dropPart(d)
					if part then
						lastGrab = os.clock()
						local back = hrp.CFrame
						hrp.CFrame = part.CFrame + Vector3.new(0, 2, 0)
						if firetouchinterest then
							pcall(firetouchinterest, hrp, part, 0)
							task.wait(0.05)
							pcall(firetouchinterest, hrp, part, 1)
						end
						task.wait(0.35)
						local _, _, hrp2 = getChar(lp)
						if hrp2 then hrp2.CFrame = back end
						break
					end
				end
			end
		end
		task.wait(0.3)
	end
end)

track(RunService.Heartbeat:Connect(function()
	local _, hum, hrp = getChar(lp)
	if not hrp then return end
	if state.fly then
		local vert = (state.up and 1 or 0) - (state.down and 1 or 0)
		hrp.AssemblyLinearVelocity = hum.MoveDirection * CONFIG.FlySpeed + Vector3.new(0, vert * CONFIG.FlySpeed, 0)
	end
	if state.pull and selected then
		local _, _, thrp = getChar(selected)
		if thrp then
			thrp.CFrame = hrp.CFrame * CONFIG.PullOffset
			thrp.AssemblyLinearVelocity = Vector3.zero
		end
	end
end))

local function makeDraggable(handle, targetFrame, onTap)
	local dragging, moved, startPos, origin, startInput = false, false, nil, nil, nil
	track(handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging, moved, startPos, origin, startInput = true, false, i.Position, targetFrame.Position, i
			i.Changed:Connect(function()
				if i.UserInputState == Enum.UserInputState.End then
					dragging = false
					if not moved and onTap then onTap() end
				end
			end)
		end
	end))
	track(UIS.InputChanged:Connect(function(i)
		if not dragging then return end
		local isMove = i.UserInputType == Enum.UserInputType.MouseMovement
		local isTouch = i.UserInputType == Enum.UserInputType.Touch and i == startInput
		if isMove or isTouch then
			local d = i.Position - startPos
			if d.Magnitude > 6 then moved = true end
			if moved then
				targetFrame.Position = UDim2.new(origin.X.Scale, origin.X.Offset + d.X, origin.Y.Scale, origin.Y.Offset + d.Y)
			end
		end
	end))
end

	corner(mark, 3)

	lbl.Position = UDim2.fromOffset(28, 0)

	lbl.Font = Enum.Font.GothamBold

		state[key] = v

	track(row.Activated:Connect(function() set(not state[key]) end))
